# services/common: knsvc (no database code) + knsvc_db (store layer + drivers).
# See services/CMakeLists.txt.

set(KNSVC_CORE_SOURCES
    src/buf.c src/log.c src/json.c src/config.c src/loop.c
    src/http.c src/http_parse.c src/url.c
    src/ws.c src/ws_frame.c src/crypto.c src/msgpack.c)

# Only files that exist are compiled, so a module or test that is being added
# never breaks configuring the whole repository.
function(knsvc_existing out)
    set(list "")
    foreach (f IN LISTS ARGN)
        if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${f})
            list(APPEND list ${f})
        else ()
            message(STATUS "services/common: ${f} missing, skipped")
        endif ()
    endforeach ()
    set(${out} ${list} PARENT_SCOPE)
endfunction ()

knsvc_existing(KNSVC_CORE_SOURCES ${KNSVC_CORE_SOURCES})
add_library(knsvc STATIC ${KNSVC_CORE_SOURCES})
target_include_directories(knsvc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(knsvc PUBLIC cJSON OpenSSL::Crypto m PRIVATE Threads::Threads)
# wss:// client (ws.c) when libssl is there: game servers reach the coordinator through
# Cloudflare (https only). Without it the ws client is ws:// only.
if (TARGET OpenSSL::SSL)
    target_link_libraries(knsvc PUBLIC OpenSSL::SSL)
    target_compile_definitions(knsvc PRIVATE KNSVC_HAVE_TLS=1)
else ()
    message(STATUS "services/common: OpenSSL libssl not found; the knsvc ws client is ws:// only")
endif ()
target_compile_options(knsvc PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_compile_definitions(knsvc PRIVATE _GNU_SOURCE)
# Optional crypt(3) (yescrypt/sha512crypt) verification in kn_password_verify.
find_library(KN_LIBCRYPT crypt)
find_path(KN_CRYPT_H crypt.h)
if (KN_LIBCRYPT AND KN_CRYPT_H)
    target_link_libraries(knsvc PRIVATE ${KN_LIBCRYPT})
    target_compile_definitions(knsvc PRIVATE KNSVC_HAVE_LIBCRYPT=1)
endif ()

if (KNSVC_DB_AVAILABLE)
    knsvc_existing(KNSVC_DB_SOURCES src/store.c src/store_result.c src/store_sql.c src/store_worker.c
                   src/store_drivers.c src/migrate.c)
    add_library(knsvc_db STATIC ${KNSVC_DB_SOURCES})
    target_link_libraries(knsvc_db PUBLIC knsvc Threads::Threads)
    target_compile_options(knsvc_db PRIVATE -Wall -Wextra -Wno-unused-parameter)
    target_compile_definitions(knsvc_db PRIVATE _GNU_SOURCE
        KN_SOURCE_MIGRATIONS_DIR="${CMAKE_SOURCE_DIR}/services/db/migrations")

    # Store drivers (knsvc/store.h), one registration line each:
    #   knsvc_store_driver(<OPTION> <name> <source> <client library targets...>)
    # builds <source> (which defines `kn_store_driver_<name>`) when KNSVC_STORE_<OPTION>
    # is ON (option + find_package in services/CMakeLists.txt) and lists it in the
    # generated knsvc_store_drivers.h that src/store_drivers.c includes.
    set(KNSVC_STORE_DRIVER_NAMES "")
    function(knsvc_store_driver option name source)
        if (KNSVC_STORE_${option})
            target_sources(knsvc_db PRIVATE ${source})
            target_link_libraries(knsvc_db PUBLIC ${ARGN})
            target_compile_definitions(knsvc_db PRIVATE KNSVC_STORE_HAVE_${option}=1)
            set(KNSVC_STORE_DRIVER_NAMES ${KNSVC_STORE_DRIVER_NAMES} ${name} PARENT_SCOPE)
        endif ()
    endfunction ()
    knsvc_store_driver(POSTGRES postgres src/drivers/store_postgres.c ${KN_LIBPQ_TARGET})
    knsvc_store_driver(SQLITE sqlite src/drivers/store_sqlite.c ${KN_SQLITE_TARGET})
    knsvc_store_driver(MYSQL mysql src/drivers/store_mysql.c ${KN_MYSQL_TARGET})
    set(_kn_drivers_h "/* generated by services/common/CMakeLists.txt (knsvc_store_driver) */\n")
    foreach (n IN LISTS KNSVC_STORE_DRIVER_NAMES)
        string(APPEND _kn_drivers_h "KN_STORE_DRIVER(${n})\n")
    endforeach ()
    file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/knsvc_store_drivers.h CONTENT "${_kn_drivers_h}")
    target_include_directories(knsvc_db PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

    if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/kn_migrate.c)
        add_executable(kn_migrate tools/kn_migrate.c)
        target_link_libraries(kn_migrate PRIVATE knsvc_db)
    endif ()
endif ()

# ---- tests (ctest; DB tests skip with code 77 when PostgreSQL is unreachable)
set(KNSVC_TESTS buf config crypto msgpack loop http ws)
foreach (t IN LISTS KNSVC_TESTS)
    if (NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${t}.c)
        continue()
    endif ()
    add_executable(knsvc_test_${t} tests/test_${t}.c)
    target_link_libraries(knsvc_test_${t} PRIVATE knsvc Threads::Threads)
    target_compile_definitions(knsvc_test_${t} PRIVATE _GNU_SOURCE
        KN_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" KN_REPO_ROOT="${CMAKE_SOURCE_DIR}")
    add_test(NAME knsvc_${t} COMMAND knsvc_test_${t})
    set_tests_properties(knsvc_${t} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 LABELS knsvc)
endforeach ()

    target_link_libraries(knsvc_test_store PRIVATE knsvc_db Threads::Threads)

        set_tests_properties(knsvc_store_${d} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 LABELS "knsvc;db;${d}")
    endforeach ()
endif ()

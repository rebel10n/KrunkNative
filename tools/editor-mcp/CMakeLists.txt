# kn_editor_mcp: the Krunker map editor MCP server (tools/editor-mcp/README.md).
# Pure C11 + the shared game code (map parsing, catalogue, bots / nav grid, editor bus) and
# cJSON. No GL: renders run client_bin as a subprocess. Added by the root CMakeLists.txt
# (native builds); built by `make`, tested by `make editor-mcp-test` (part of `make check`).

# POSIX only (poll, AF_UNIX/arpa/inet, fork/exec of client_bin, realpath, mkdtemp): not built
# on Windows (MinGW), where the release build would otherwise fail on it.
if (WIN32)
    message(STATUS "kn_editor_mcp: POSIX only, not built on Windows")
    return()
endif ()

# data/schema.json (generated by gen/gen_schema.py) embedded as kn_schema_json[]
set(KN_MCP_SCHEMA ${CMAKE_CURRENT_SOURCE_DIR}/data/schema.json)
set(KN_MCP_SCHEMA_C ${CMAKE_CURRENT_BINARY_DIR}/schema_embed.c)
add_custom_command(
    OUTPUT ${KN_MCP_SCHEMA_C}
    COMMAND ${CMAKE_COMMAND} -DIN=${KN_MCP_SCHEMA} -DOUT=${KN_MCP_SCHEMA_C} -P ${CMAKE_CURRENT_SOURCE_DIR}/embed.cmake
    DEPENDS ${KN_MCP_SCHEMA} ${CMAKE_CURRENT_SOURCE_DIR}/embed.cmake
    COMMENT "Embedding the kn_editor_mcp schema"
    VERBATIM)

file(GLOB KN_MCP_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
file(GLOB KN_MCP_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h)

# KrunkScript (doc/v7/krunkscript.md): the front end in shared/src/ks comes with SHARED_SOURCES;
# src/ks_bridge.c reaches ks_validate through weak references, so the server still links (and
# reports "compiler not linked") in a tree without it.
add_executable(kn_editor_mcp ${KN_MCP_SOURCES} ${KN_MCP_HEADERS} ${KN_MCP_SCHEMA_C} ${SHARED_SOURCES} ${SHARED_HEADERS})
if (NOT MSVC)
    set_source_files_properties(${KN_MCP_SOURCES} PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wno-unused-parameter")
endif ()
target_include_directories(kn_editor_mcp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/shared/include)
target_link_libraries(kn_editor_mcp PRIVATE cJSON pcg_basic m)
if (CMAKE_SYSTEM_NAME STREQUAL "Windows")
    target_link_libraries(kn_editor_mcp PRIVATE ws2_32)
endif ()
